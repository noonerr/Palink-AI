let toastT;
const log = (msg, isErr=false) => { console.log('[Sys]', msg); const t=document.getElementById('toast'); t.innerText=msg; t.className = isErr ? 'show err' : 'show'; clearTimeout(toastT); toastT=setTimeout(()=>t.className='',3000); };

const UI = {
    toggleTheme: () => {
        const body = document.body; const btn = document.getElementById('theme-btn');
        if (body.getAttribute('data-theme') === 'light') { body.removeAttribute('data-theme'); btn.innerText = '🌙'; localStorage.setItem('theme', 'dark'); } 
        else { body.setAttribute('data-theme', 'light'); btn.innerText = '🌞'; localStorage.setItem('theme', 'light'); }
    },
    initTheme: () => { if (localStorage.getItem('theme') === 'light') { document.body.setAttribute('data-theme', 'light'); document.getElementById('theme-btn').innerText = '🌞'; } },
    showHelp: () => { document.getElementById('help-modal').style.display = 'flex'; setTimeout(()=>document.querySelector('.modal').classList.add('show'),10); },
    hideHelp: () => { document.querySelector('.modal').classList.remove('show'); setTimeout(()=>document.getElementById('help-modal').style.display = 'none',200); },
    go: (id) => { document.querySelectorAll('.view').forEach(e=>e.classList.remove('active')); document.getElementById(id).classList.add('active'); },
    reset: () => { window.history.replaceState(null, null, window.location.pathname); location.reload(); },
    toSender: () => UI.go('v-send-1'),
    toReceiver: () => { UI.go('v-recv'); setTimeout(()=>document.getElementById('in-code').focus(),200); },
    setNetState: (s) => { 
        const el=document.getElementById('net-badge'), tx=document.getElementById('net-txt'); el.className='status-pill'; 
        if(s==='online') { el.classList.add('online'); tx.innerText='服务器在线'; }
        else if(s==='p2p') { el.classList.add('p2p'); tx.innerText='P2P 直连'; }
        else if(s==='cft') { el.classList.add('cft'); tx.innerText='CFT 隧道中'; }
        else if(s==='offline') { el.classList.add('offline'); tx.innerText='未连接/断开'; }
        else { tx.innerText='连接中...'; } 
    },
    updateProgress: (pct, speed, mode) => {
        const offset = 527 - (pct/100)*527; document.getElementById('ring').style.strokeDashoffset = offset;
        document.getElementById('t-pct').innerText = Math.floor(pct) + '%'; document.getElementById('t-speed').innerText = speed;
        if (mode === '中转' && (location.hostname.includes('cloudflare') || location.hostname.includes('cf'))) { mode = 'CFT 隧道'; UI.setNetState('cft'); }
        document.getElementById('t-status').innerText = `模式: ${mode}`;
        if(pct >= 100) { document.getElementById('t-status').innerText = '传输完成'; document.getElementById('ring').style.stroke = '#00ff88'; document.getElementById('btn-done').style.display = 'block'; }
    },

const Core = {
    socket: null, roomId: null, file: null, peer: null, dc: null, isP2P: false, isInitiator: false,
    CHUNK_SIZE: 512 * 1024, // 512KB 分片
    BATCH_SIZE: 8,          // 8 线程并发
    offset: 0, startTime: 0, recvBuffer: [], recvSize: 0, meta: null, 
    ackResolvers: [], // ACK 队列
    stopFlag: false, lastUiTime: 0,

    init: () => {
        const proto = location.protocol==='https:'?'wss':'ws'; Core.ws = new WebSocket(`${proto}://${location.host}/ws`); Core.ws.binaryType = 'arraybuffer';
        Core.ws.onopen = () => { UI.setNetState('online'); log('连接服务器成功'); };
        Core.ws.onclose = () => { UI.setNetState('offline'); log('服务器连接断开', true); };
        Core.ws.onmessage = (e) => { if(Core.stopFlag) return; if(e.data instanceof ArrayBuffer) Core.handleBinary(e.data); else try{Core.handleJSON(JSON.parse(e.data))}catch(err){} };
        const params = new URLSearchParams(location.search); if(params.get('c')) { document.getElementById('in-code').value = params.get('c'); UI.toReceiver(); }
        document.getElementById('f-input').addEventListener('change', (e) => { if(e.target.files.length) { Core.file = e.target.files[0]; document.getElementById('f-name').innerText = Core.file.name; document.getElementById('drop-area').classList.add('active'); setTimeout(Core.createRoom, 500); } });
        setInterval(()=> { if(Core.ws.readyState===1 && !Core.stopFlag) Core.sendJSON({type:'ping'}); }, 10000);
    },
    sendJSON: (msg) => { if(Core.ws.readyState===1) Core.ws.send(JSON.stringify({roomId: Core.roomId, ...msg})); },
    handleJSON: (msg) => {
        switch(msg.type) {
            case 'joined': log(`房间 ${Core.roomId} 就绪`); break;
            case 'err': alert(msg.payload); UI.reset(); break;
            case 'peer-left': Core.stopFlag = true; UI.showError("对方已中断连接"); break;
            case 'transfer-complete': 
                if(Core.isInitiator) { 
                    Core.ackResolvers.forEach(r => r()); Core.ackResolvers = [];
                    UI.updateProgress(100, '完成', '完成'); 
                } break;
            case 'ready-to-p2p': if(Core.isInitiator){ Core.setupPeer(); setTimeout(()=>{if(!Core.sending)Core.startSending()},500); } break;
            case 'signal': Core.handleSignal(msg.payload); break;
            case 'meta': Core.onMeta(msg.payload); break;
        }
    },
    handleBinary: (buf) => {
        const view = new DataView(buf); const flag = view.getUint8(0);
        if (flag === 1) { 
            const resolve = Core.ackResolvers.shift(); 
            if(resolve) resolve(); 
        }
        else if (flag === 2) { const dec = new TextDecoder(); const text = dec.decode(buf.slice(1)); UI.addChatMessage(text, false); }
        else { Core.onChunk(buf.slice(1)); }
    },
    sendTextP2P: (text) => { const enc = new TextEncoder(); const textBuf = enc.encode(text); const packet = new Uint8Array(textBuf.byteLength + 1); packet[0] = 2; packet.set(textBuf, 1); if(Core.isP2P && Core.dc?.readyState==='open') { Core.dc.send(packet); } else if(Core.ws.readyState===1) { Core.ws.send(packet); } },
    createRoom: () => { Core.isInitiator = true; Core.roomId = Math.random().toString(36).substr(2, 4).toUpperCase(); Core.sendJSON({type: 'join', roomId: Core.roomId, role: 'sender'}); document.getElementById('show-code').innerText = Core.roomId; document.getElementById('qr-box').innerHTML = ''; new QRCode(document.getElementById('qr-box'), { text: location.origin + location.pathname + '?c=' + Core.roomId, width: 180, height: 180 }); UI.go('v-send-2'); },
    joinRoom: () => { const c = document.getElementById('in-code').value.toUpperCase(); if(c.length!==4) return alert('请输入4位口令'); Core.roomId = c; Core.sendJSON({type: 'join', roomId: Core.roomId, role: 'receiver'}); UI.go('v-transfer'); },
    copyLink: () => { navigator.clipboard.writeText(location.origin + location.pathname + '?c=' + Core.roomId); log('已复制'); },
    setupPeer: () => {
        Core.peer = new RTCPeerConnection({iceServers:[{urls:'stun:stun.miwifi.com'}, {urls:'stun:stun.l.google.com:19302'}]});
        Core.peer.onicecandidate = e => e.candidate && Core.sendJSON({type:'signal', payload:{candidate: e.candidate}});
        Core.peer.onconnectionstatechange = () => { if(Core.peer.connectionState==='connected'){ Core.isP2P=true; UI.setNetState('p2p'); } else if(Core.peer.connectionState==='disconnected'||Core.peer.connectionState==='failed') { Core.isP2P=false; UI.setNetState('online'); } };
        if(Core.isInitiator) { Core.dc = Core.peer.createDataChannel('file'); Core.dc.onopen = () => { Core.isP2P=true; UI.setNetState('p2p'); }; Core.dc.onmessage = e => Core.handleBinary(e.data); Core.createOffer(); } 
        else { Core.peer.ondatachannel = e => { Core.dc = e.channel; Core.dc.onmessage = evt => Core.handleBinary(evt.data); Core.isP2P=true; }; }
    },
    createOffer: async () => { const o=await Core.peer.createOffer(); await Core.peer.setLocalDescription(o); Core.sendJSON({type:'signal', payload:{sdp:o}}); },
    handleSignal: async (d) => { if(!Core.peer)Core.setupPeer(); if(d.sdp){await Core.peer.setRemoteDescription(d.sdp);if(d.sdp.type==='offer'){const a=await Core.peer.createAnswer();await Core.peer.setLocalDescription(a);Core.sendJSON({type:'signal', payload:{sdp:a}});}}else if(d.candidate)await Core.peer.addIceCandidate(d.candidate); },
    startSending: () => { if(Core.sending || !Core.file) return; Core.sending = true; UI.go('v-transfer'); Core.offset = 0; Core.startTime = Date.now(); UI.updateProgress(0, '0 MB/s', '准备'); Core.sendJSON({type: 'meta', payload: {name: Core.file.name, size: Core.file.size, type: Core.file.type}}); Core.waitForAck().then(Core.sendLoop); },
    
    sendLoop: async () => {
        const isSmallFile = Core.file.size < 2 * 1024 * 1024; let pending = []; 
        while(Core.offset < Core.file.size) {
            if(Core.stopFlag) return;
            while(pending.length < Core.BATCH_SIZE && Core.offset < Core.file.size) {
                const end = Math.min(Core.file.size, Core.offset + Core.CHUNK_SIZE);
                const slice = Core.file.slice(Core.offset, end);
                const buf = await slice.arrayBuffer();
                const packet = new Uint8Array(buf.byteLength + 1); packet[0] = 0; packet.set(new Uint8Array(buf), 1);
                pending.push(Core.sendPacketAndWait(packet.buffer)); Core.offset = end;
            }
            await Promise.all(pending); pending = []; 
            const now = Date.now();
            if (isSmallFile || now - Core.lastUiTime > 100) {
                const pct = (Core.offset / Core.file.size) * 100; const speed = (Core.offset / 1024 / 1024 / ((now - Core.startTime)/1000 || 0.001)).toFixed(2) + ' MB/s';
                const mode = Core.isP2P ? 'P2P' : '中转'; UI.updateProgress(pct, speed, mode); Core.lastUiTime = now;
            }
        }
        // 发送完不更新 UI，等待接收端确认
    },

    sendPacketAndWait: (buf) => {
        return new Promise((resolve) => {
            Core.ackResolvers.push(resolve);
            const trySend = () => {
                if(Core.stopFlag) return;
                try { 
                    if(Core.dc && Core.dc.readyState==='open') { Core.isP2P = true; Core.dc.send(buf); } 
                    else { Core.ws.send(buf); } 
                } catch(e) { 
                    // [修复] 移除死磕重试，只尝试降级。防止数据写入重复。
                    try { Core.ws.send(buf); } catch(err) { UI.showError('发送失败，连接中断'); Core.stopFlag = true; }
                }
            }; trySend();
        });
    },
    waitForAck: () => new Promise(r => Core.ackResolvers.push(r)),
    onMeta: (m) => { if(Core.stopFlag) return; Core.meta = m; Core.recvBuffer = []; Core.recvSize = 0; Core.startTime = Date.now(); UI.go('v-transfer'); UI.updateProgress(0, '0 MB/s', '连接中'); Core.sendAck(); },
    onChunk: (chunk) => {
        if(Core.stopFlag) return; Core.recvBuffer.push(chunk); Core.recvSize += chunk.byteLength;

        Core.sendAck();

        if (Core.recvSize < Core.meta.size) {
            if (isSmallFile || now - Core.lastUiTime > 100) {

                UI.updateProgress(pct, speed, Core.isP2P ? 'P2P' : '中转'); Core.lastUiTime = now;
            }
        } else { Core.sendJSON({type: 'transfer-complete'}); UI.updateProgress(100, '完成', '完成'); setTimeout(Core.saveFile, 100); }
    },
    sendAck: () => { const p=new Uint8Array(1); p[0]=1; if(Core.isP2P&&Core.dc?.readyState==='open')Core.dc.send(p); else Core.ws.send(p); },
    saveFile: () => { const blob = new Blob(Core.recvBuffer, { type: Core.meta.type }); const url = URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = Core.meta.name; a.click(); URL.revokeObjectURL(url); Core.recvBuffer = []; },
    initAudioHack: () => { if(Core.audioCtx)return; const A=window.AudioContext||window.webkitAudioContext; if(!A)return; Core.audioCtx=new A(); const o=Core.audioCtx.createOscillator(); const g=Core.audioCtx.createGain(); o.type='sine'; o.frequency.value=0.01; g.gain.value=0.0001; o.connect(g); g.connect(Core.audioCtx.destination); o.start(); }
};
