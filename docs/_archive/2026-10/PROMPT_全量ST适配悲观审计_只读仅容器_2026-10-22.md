### 3.2 档 A：把真实提示词 dump 出来（脚本可直接用）

```python
# .tmp_audit/dump_prompt.py —— 只读：不 commit、不 flush、不写库
import asyncio, json, sys, os
sys.path.insert(0, "/app")
from app.database import SessionLocal
from app.models import Character, CharacterChatSession, User
from app.api.character_ext import (
    _build_char_system_prompt, _replace_placeholders, _get_full_branch_history,
    _get_ancestor_branch_ids, _get_ancestor_branch_scopes, _contains_chinese,
    _apply_plugin_regex_scripts, _apply_regex_scripts, _apply_prompt_regex_to_messages,
)
from app.services.roleplay_prompt_assembly import (
    PromptAssemblyRequest, PromptAssemblyDeps, assemble_roleplay_prompt,
)

CHAR_ID = os.environ.get("AUDIT_CHAR_ID", "c28061bb-a9b7-46a9-a255-2692d57001fc")
SESSION_ID = os.environ.get("AUDIT_SESSION_ID", "")   # 空 = 取该卡最近会话
USER_MSG = os.environ.get("AUDIT_USER_MSG", "继续")
MODEL = os.environ.get("AUDIT_MODEL", "deepseek-v4-flash")
OUT = os.environ.get("AUDIT_OUT", "/tmp/prompt_dump.json")

async def main():
    db = SessionLocal()
    try:
        char = db.query(Character).filter(Character.id == CHAR_ID).first()
        assert char, "character not found"
        q = db.query(CharacterChatSession).filter(CharacterChatSession.character_id == CHAR_ID)
        sess = (q.filter(CharacterChatSession.id == SESSION_ID).first() if SESSION_ID
                else q.order_by(CharacterChatSession.updated_at.desc()).first())
        assert sess, "session not found"
        user = db.query(User).filter(User.id == sess.user_id).first()
        req = PromptAssemblyRequest(
            db=db, user=user, char=char, session_id=sess.id, branch_id="",
            message=USER_MSG, images=[], model=MODEL,
            user_nickname=(user.username or "用户"), response_length=None, max_tokens=2048,
        )
        deps = PromptAssemblyDeps(
            build_system_prompt=_build_char_system_prompt,
            replace_placeholders=_replace_placeholders,
            get_full_branch_history=_get_full_branch_history,
            get_ancestor_branch_ids=_get_ancestor_branch_ids,
            get_ancestor_branch_scopes=_get_ancestor_branch_scopes,
            contains_chinese=_contains_chinese,
            apply_plugin_regex_scripts=_apply_plugin_regex_scripts,
            apply_regex_scripts=_apply_regex_scripts,
            apply_prompt_regex_to_messages=_apply_prompt_regex_to_messages,
        )
        res = await assemble_roleplay_prompt(req, deps)
        payload = {
            "char_id": CHAR_ID, "session_id": sess.id, "model": MODEL,
            "message_count": len(res.messages),
            "messages": [{"role": m.get("role"), "content": m.get("content")} for m in res.messages],
            "debug": res.debug_dict(),
            "dynamic_context_parts": getattr(res, "dynamic_context_parts", []),
            "effective_max_tokens": getattr(res, "effective_max_tokens", None),
            "token_budget": getattr(res, "token_budget", None),
            "total_tokens_estimate": getattr(res, "total_tokens_estimate", None),
        }
        with open(OUT, "w", encoding="utf-8") as f:
            json.dump(payload, f, ensure_ascii=False, indent=1)
        print("OK", OUT, "msgs=", len(res.messages))
    finally:
        db.close()

asyncio.run(main())
```

```js
// .tmp_audit/mock_provider.js —— 用法：page.route 前置注册，先于任何 goto
const FAKE = process.env.FAKE_REPLY || '【测试锚点-A】……（子 agent 产出的文本）';

async function installMockProvider(page) {
  const hit = [];
  const sse = (obj) => `data: ${JSON.stringify(obj)}\n\n`;
  const patterns = [
    '**/api/character-chat',            // 主链（SSE）
    '**/api/character-chat/smart-card-generate',
    '**/api/chats/generate-raw',
    '**/api/sessions/*/regenerate',
    '**/api/sessions/*/continue',
    '**/api/sessions/*/mvu-secondary',
    '**/api/backends/chat-completions/generate',
    '**/v1/chat/completions',           // 兜底：任何直连 provider
  ];
  for (const p of patterns) {
    await page.route(p, async (route) => {
      if (route.request().method() !== 'POST') return route.continue();
      hit.push(p);
      const chunks = FAKE.match(/[\s\S]{1,12}/g) || [''];
      let body = sse({ session_id: 'mock-session', branch_id: 'mock-branch' });
      for (const c of chunks) body += sse({ content: c });
      body += sse({ type: 'final_content', content: FAKE, message_id: 'mock-msg-1' });
      body += 'data: [DONE]\n\n';
      await route.fulfill({
        status: 200,
        headers: { 'content-type': 'text/event-stream; charset=utf-8', 'cache-control': 'no-cache' },
        body,
      });
    });
  }
  return hit;
}
```

**第三步：拦截命中率必须是 100%。** 跑完断言：
- `hit.length > 0`（至少真拦到一次，否则说明你的 pattern 写错了、请求走了别的路径）；
- `page.on('request')` 里**没有任何**请求指向非 `localhost`/非站点主机的域名；
- 后端容器日志（`docker logs --since 10m palink-ai-backend-1`）里**没有** provider 调用记录、没有 402/429 余额错误。

### 3.5 档 C：契约 mock（前端单测）

只验"组件拿到某个 SSE 序列后怎么表现"，不验后端。已有先例：`frontend/src/**/__tests__/*.test.tsx` 里大量 `vi.mock`。
要求：mock 的 fixture 必须是**从档 A 真实 dump 里裁剪出来的形状**，不许手捏一个"理想形状"（否则测的是你的想象）。

### 3.6 反作弊：怎么证明"真的没打网关"

三层证据，缺一层就只能写"未完全证明"：

1. **物理层**：playwright 容器 `--dns 127.0.0.1`（无 DNS 服务 ⇒ 外部域名解析必失败），站点用容器 IP 直连（`docker network inspect palink-ai_frontend-backend` 取 frontend 的 IP）。
2. **网络层**：`page.on('request')` 全量记录 + 断言 host 白名单。
3. **服务端层**：`docker logs` 无 provider 出站；DB 无新增 assistant 消息（G5 的计数不变）。

## 6. 数据真源与陷阱核对（**每个"明明改了没生效"都先查这里**）
