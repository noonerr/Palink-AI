// --- 基础组件 (Shared Components) ---

const ThinkingProcess = ({ content, streaming }) => {
    const [isExpanded, setIsExpanded] = useState(false);
    useEffect(() => { if (streaming && content) setIsExpanded(true); }, [streaming]);
    if (!content) return null;
    return (
        <div className="mb-4 rounded-xl border border-gray-200 dark:border-gray-700 bg-gray-50/50 dark:bg-[#252525]/50 overflow-hidden">
            <button onClick={() => setIsExpanded(!isExpanded)} className="w-full flex items-center gap-2 px-4 py-2.5 text-xs font-bold text-gray-500 dark:text-gray-400 hover:bg-gray-100 dark:hover:bg-[#333] transition-colors">
                <Brain size={14} className={streaming ? "animate-pulse text-indigo-500" : ""}/><span>深度思考过程</span>
                <span className="ml-auto text-xs opacity-50">{isExpanded ? '收起' : '展开'}</span>
                {isExpanded ? <ChevronUp size={14}/> : <ChevronDown size={14}/>}
            </button>
            {isExpanded && (
                <div className="px-4 py-3 text-sm text-gray-600 dark:text-gray-300 font-mono border-t border-gray-200 dark:border-gray-700 animate-in slide-in-from-top-2 fade-in duration-200">
                    <ReactMarkdown components={{code:CodeBlock}}>{content}</ReactMarkdown>
                </div>
            )}
        </div>
    );
};

const ChevronUp = ({size, className}) => <ChevronDown size={size} className={`rotate-180 ${className}`} />;

const CodeBlock = memo(({ inline, className, children, ...props }) => {
  const match = /language-(\w+)/.exec(className || '');
  const [copied, setCopied] = useState(false);
  if (inline) return <code className="bg-gray-100 dark:bg-gray-800 px-1.5 py-0.5 rounded-lg text-sm text-pink-500 font-mono" {...props}>{children}</code>;
  return (
    <div className="my-4 rounded-2xl overflow-hidden border border-gray-200 dark:border-gray-700 bg-[#1e1e1e] shadow-sm">
      <div className="flex justify-between px-4 py-2 bg-[#2d2d2d] text-xs text-gray-400 items-center">
        <span className="font-mono">{match?.[1] || 'text'}</span>
        <button onClick={()=>{navigator.clipboard.writeText(String(children));setCopied(true);setTimeout(()=>setCopied(false),2000)}} className="hover:text-white flex items-center gap-1 transition-colors">
          {copied ? <Check size={12}/> : <Copy size={12}/>} {copied ? '已复制' : '复制代码'}
        </button>
      </div>
      <SyntaxHighlighter style={vscDarkPlus} language={match?.[1]} PreTag="div" customStyle={{margin:0, padding:'1.25rem', fontSize:'0.9em', lineHeight:'1.6'}} {...props}>
        {String(children).replace(/\n$/, '')}
      </SyntaxHighlighter>
    </div>
  );
});

const Avatar = memo(({ url, username, size="md", className="" }) => {
  const sizeClass = { sm: 'w-8 h-8 text-xs', md: 'w-10 h-10 text-sm', lg: 'w-16 h-16 text-2xl', xl: 'w-24 h-24 text-4xl' }[size] || 'w-10 h-10 text-sm';
  if (url) return <img src={url} className={`${sizeClass} rounded-full object-cover border border-gray-200 dark:border-gray-700 shadow-sm bg-white dark:bg-gray-800 ${className}`} alt="Avatar" />;
  return <div className={`${sizeClass} rounded-full bg-gradient-to-br from-indigo-500 to-purple-600 text-white flex items-center justify-center font-bold uppercase shadow-sm ${className}`}>{username ? username[0] : 'U'}</div>;
});

// --- View 1: Chat View (Stream Logic + Sidebar) ---
const ChatView = ({ token, user, filesToUpload, onClearUploads }) => {
  const [sessions, setSessions] = useState([]);
  const [activeSid, setActiveSid] = useState(null);
  const [messages, setMessages] = useState([]);
  const [input, setInput] = useState('');
  const [streaming, setStreaming] = useState(false);
  const [models, setModels] = useState([]);
  const [curModel, setCurModel] = useState('');
  const [sidebarOpen, setSidebarOpen] = useState(true);
  const [attachments, setAttachments] = useState([]);
  const [suggestions, setSuggestions] = useState([]); 
  const [starters, setStarters] = useState([]);
  const [contextInfo, setContextInfo] = useState(null); 
  const [loadingStarters, setLoadingStarters] = useState(true);
  
  const bottomRef = useRef(null);
  const abortCtrl = useRef(null);
  const imgInputRef = useRef(null);
  const fileInputRef = useRef(null);

  // 处理拖拽进来的文件
  useEffect(() => {
      if (filesToUpload && filesToUpload.length > 0) {
          // 模拟 Event 对象结构进行上传
          filesToUpload.forEach(file => {
             const mockEvent = { target: { files: [file] } };
             handleUpload(mockEvent, file.type.startsWith('image') ? 'image' : 'file');
          });
          onClearUploads();
      }
  }, [filesToUpload]);

  useEffect(() => {
     fetch('/api/models').then(r=>r.json()).then(d => { setModels(d); if(d.length && !curModel) setCurModel(d[0].id); });
     refreshSessions();
     fetch('/api/recommendations/starters').then(r=>r.json()).then(s => { setStarters(s); setLoadingStarters(false); }).catch(()=>setLoadingStarters(false));
  }, [user]);

  useEffect(() => {
      if(activeSid) {
          fetch(`/api/sessions/${activeSid}/messages`, { headers: { Authorization: `Bearer ${token}` } })
            .then(r=>r.json()).then(data => { setMessages(data); setSuggestions([]); setContextInfo(null); });
      } else { setMessages([]); setSuggestions([]); }
  }, [activeSid]);

  useEffect(() => { bottomRef.current?.scrollIntoView({ behavior: 'smooth' }); }, [messages, streaming]);
  
  const refreshSessions = useCallback(() => { fetch('/api/sessions', { headers: { Authorization: `Bearer ${token}` } }).then(r=>r.json()).then(setSessions); }, [token]);
  const createNewChat = () => { setActiveSid(null); setMessages([]); setAttachments([]); setSuggestions([]); };

  const handleUpload = async (e, type) => {
      const file = e.target.files?.[0]; if (!file) return;
      const reader = new FileReader();
      reader.onload = async (event) => {
          try {
              const res = await fetch('/api/upload', { method: 'POST', headers: { 'Content-Type': 'application/json', Authorization: `Bearer ${token}` }, body: JSON.stringify({ filename: file.name, data: event.target.result }) });
              if(res.ok) { const data = await res.json(); setAttachments(prev => [...prev, { type, url: data.url, name: file.name }]); }
          } catch(e) {}
      }; reader.readAsDataURL(file); if(e.target.value) e.target.value = '';
  };

  const removeAttachment = (idx) => setAttachments(prev => prev.filter((_, i) => i !== idx));

  const sendMessage = async (overrideText) => {
    const txt = overrideText || input; if((!txt.trim() && attachments.length === 0) || streaming) return;
    setSuggestions([]); setInput(''); setAttachments([]); setStreaming(true);
    
    let displayContent = txt;
    if (attachments.length > 0) { displayContent += (txt ? "\n" : ""); attachments.forEach(att => { displayContent += `<file-card type="${att.type}" name="${att.name}" url="${att.url}" />`; }); }

    setMessages(prev => [...prev, {role:'user', content:displayContent}, {role:'assistant', content:''}]);
    abortCtrl.current = new AbortController();
    let fullReasoning = '', fullContent = '', currentSid = activeSid;

    try {
      const res = await fetch('/api/chat', { method: 'POST', headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ session_id: currentSid, message: txt, model: curModel, images: attachments.filter(a => a.type === 'image').map(a => a.url), files: attachments.filter(a => a.type === 'file').map(a => a.url) }), signal: abortCtrl.current.signal });
      const reader = res.body.getReader(); const dec = new TextDecoder();
      while(true) {
         const {done, value} = await reader.read(); if(done) break;
         const lines = dec.decode(value, {stream:true}).split('\n');
         for(const line of lines) {
            if(line.startsWith('data: ')) {
               const d = line.slice(6); if(d === '[DONE]') continue;
               try {
                  const j = JSON.parse(d);
                  if(j.type === 'meta') { setContextInfo({ remaining: j.remaining_tokens, max: j.max_tokens }); continue; }
                  if (j.reasoning) fullReasoning += j.reasoning;
                  if (j.content) fullContent += j.content;
                  setMessages(prev => { const n = [...prev]; n[n.length-1].content = fullReasoning ? `<think>${fullReasoning}</think>${fullContent}` : fullContent; return n; });
               } catch(e){}
            }
         }
      }
      refreshSessions();
      if(fullContent.length > 10) { fetch('/api/chat/suggestions', { method: 'POST', headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ message: fullContent, model: curModel }) }).then(r=>r.json()).then(s => setSuggestions(s)).catch(()=>{}); }
    } catch(e) { if(e.name !== 'AbortError') setMessages(prev => { const n=[...prev]; n[n.length-1].content += `\n[Error: ${e.message}]`; return n; }); } finally { setStreaming(false); abortCtrl.current = null; }
  };

  const isNewChat = !activeSid && messages.length === 0;

  return (
    <div className="flex h-full w-full bg-white dark:bg-[#121212]">
       {/* 历史侧边栏 */}
       <div className={`${sidebarOpen?'w-64':'w-0'} bg-gray-50 dark:bg-[#0f0f0f] border-r border-gray-200 dark:border-gray-800 flex flex-col transition-all duration-300 overflow-hidden shrink-0`}>
          <div className="p-4 flex items-center justify-between"><span className="text-sm font-bold text-gray-500 px-1">历史记录</span><button onClick={createNewChat} className="p-2 hover:bg-gray-200 dark:hover:bg-[#252525] rounded-xl text-gray-600 dark:text-gray-300"><MessageSquarePlus size={18}/></button></div>
          <div className="flex-1 overflow-y-auto px-2 space-y-1 scrollbar-thin pb-4">
            {sessions.map(s => <div key={s.id} onClick={()=>setActiveSid(s.id)} className={`p-3 rounded-xl text-sm cursor-pointer truncate transition-all ${activeSid===s.id ?'bg-white dark:bg-[#1e1e1e] shadow-sm font-medium dark:text-white':'text-gray-500 hover:bg-gray-200/50 dark:hover:bg-[#1a1a1a]'}`}>{s.title || 'New Chat'}</div>)}
          </div>
       </div>

       {/* 主对话区 */}
       <div className="flex-1 flex flex-col min-w-0 relative h-full">
          {/* 顶部栏 */}
          <div className="h-16 flex items-center justify-between px-6 border-b border-gray-100 dark:border-gray-800 bg-white/80 dark:bg-[#121212]/80 backdrop-blur-md z-10">
             <div className="flex items-center gap-3">
                 <button onClick={()=>setSidebarOpen(!sidebarOpen)} className="p-2 text-gray-400 hover:bg-gray-100 dark:hover:bg-[#252525] rounded-xl"><PanelLeftClose size={18}/></button>
                 <div className="w-48">
                    <select value={curModel} onChange={(e)=>setCurModel(e.target.value)} className="w-full bg-transparent text-sm font-bold outline-none dark:text-white appearance-none cursor-pointer">
                        {models.map(m=><option key={m.id} value={m.id}>{m.name}</option>)}
                    </select>
                 </div>
                 {contextInfo && <span className="text-[10px] bg-gray-100 dark:bg-[#252525] px-2 py-1 rounded text-gray-500">{contextInfo.remaining} toks left</span>}
             </div>
             <Avatar url={user.avatar} size="sm" />
          </div>

          <div className="flex-1 overflow-y-auto p-4 md:p-8 scrollbar-thin">
             {isNewChat ? (
                <div className="h-full flex flex-col items-center justify-center animate-in fade-in zoom-in duration-500">
                   <div className="w-full max-w-2xl flex flex-col items-center gap-6">
                      <div className="w-16 h-16 rounded-2xl bg-indigo-500 text-white flex items-center justify-center shadow-lg shadow-indigo-500/30 mb-2"><Bot size={32}/></div>
                      <h2 className="text-2xl font-bold dark:text-white">今天有什么可以帮您？</h2>
                      <div className="grid grid-cols-2 gap-3 w-full">
                         {loadingStarters ? <div className="col-span-2 text-center text-gray-400 text-sm">Loading...</div> : starters.map((t, i) => (
                            <button key={i} onClick={()=>sendMessage(t)} className="px-4 py-3 bg-gray-50 dark:bg-[#1e1e1e] rounded-xl text-sm font-medium text-gray-600 dark:text-gray-400 hover:bg-indigo-50 dark:hover:bg-indigo-900/20 hover:text-indigo-600 dark:hover:text-indigo-400 transition-colors text-left flex items-center justify-between group">{t} <CornerDownRight size={14} className="opacity-0 group-hover:opacity-100 transition-opacity"/></button>
                         ))}
                      </div>
                   </div>
                </div>
             ) : (
                <div className="max-w-4xl mx-auto w-full space-y-6">
                   {messages.map((m,i) => <ChatMessageItem key={i} role={m.role} content={m.content} user={user} streaming={streaming && i === messages.length - 1} />)}
                   {!streaming && suggestions.length > 0 && (
                       <div className="flex gap-2 ml-14 flex-wrap">
                           {suggestions.map((s, idx) => (
                               <button key={idx} onClick={()=>sendMessage(s)} className="text-xs bg-gray-100 dark:bg-[#252525] hover:bg-indigo-50 dark:hover:bg-indigo-900/30 text-gray-600 dark:text-gray-300 hover:text-indigo-600 px-3 py-1.5 rounded-full border border-gray-200 dark:border-gray-700 transition-colors flex items-center gap-1">{s} <Plus size={12}/></button>
                           ))}
                       </div>
                   )}
                   <div ref={bottomRef}/>
                </div>
             )}
          </div>

          {/* 输入框区域 */}
          <div className="p-6 bg-white dark:bg-[#121212]">
             <div className="max-w-4xl mx-auto relative w-full">
               {attachments.length > 0 && (
                 <div className="flex gap-2 mb-2 overflow-x-auto pb-2">
                    {attachments.map((att, idx) => (
                       <div key={idx} className="relative group bg-gray-100 dark:bg-[#252525] rounded-xl p-2 flex items-center gap-2 border dark:border-gray-700 shrink-0">
                          <span className="text-xs truncate max-w-[100px] dark:text-gray-300">{att.name}</span>
                          <button onClick={() => removeAttachment(idx)} className="hover:text-red-500"><X size={12}/></button>
                       </div>
                    ))}
                 </div>
               )}
               <div className={`flex items-center gap-2 p-2 rounded-2xl border transition-all ring-offset-2 bg-gray-50 dark:bg-[#1e1e1e] border-gray-200 dark:border-gray-700 focus-within:ring-2 focus-within:ring-indigo-500/20 focus-within:border-indigo-500`}>
                 <button onClick={()=>fileInputRef.current?.click()} className="p-2 rounded-xl text-gray-400 hover:text-indigo-600 hover:bg-indigo-50 dark:hover:bg-[#333] transition-colors"><Plus size={20}/></button>
                 <input 
                    value={input} 
                    onChange={e=>setInput(e.target.value)} 
                    onKeyDown={e=>{if(e.key==='Enter'&&!e.shiftKey){e.preventDefault();sendMessage();}}} 
                    placeholder="发送消息，或拖拽文件..." 
                    className="flex-1 bg-transparent border-none outline-none text-sm px-2 max-h-32 dark:text-white"
                 />
                 <div className="flex gap-1">
                    <button onClick={()=>imgInputRef.current?.click()} className="p-2 text-gray-400 hover:text-gray-600"><ImageIcon size={20}/></button>
                    <button onClick={()=>sendMessage()} disabled={!input.trim()&&!attachments.length} className={`p-2 rounded-xl transition-all ${input.trim()||attachments.length ? 'bg-black dark:bg-white text-white dark:text-black shadow-md' : 'bg-gray-200 dark:bg-[#333] text-gray-400 cursor-not-allowed'}`}><ArrowUpCircle size={20}/></button>
                 </div>
               </div>
               <p className="text-center text-xs text-gray-400 mt-2">支持全屏拖拽上传 • AI 模型由 {models.find(m=>m.id===curModel)?.provider || 'System'} 提供</p>
             </div>
          </div>
          <input type="file" ref={imgInputRef} className="hidden" accept="image/*" onChange={(e)=>handleUpload(e, 'image')} />
          <input type="file" ref={fileInputRef} className="hidden" accept=".txt,.md,.py,.js,.json,.csv,.html,.css,.pdf" onChange={(e)=>handleUpload(e, 'file')} />
       </div>
    </div>
  );
};

// --- View 2: Workspace View (Grid/List + Breadcrumbs) ---
const WorkspaceView = ({ token, filesToUpload, onClearUploads }) => {
    const [path, setPath] = useState([]); 
    const [items, setItems] = useState({ folders: [], files: [], usage: 0, limit: 0 });
    const [loading, setLoading] = useState(false);
    const [viewMode, setViewMode] = useState('grid');
    const [isCreateFolder, setIsCreateFolder] = useState(false);
    const [newFolderName, setNewFolderName] = useState("");
    const fileInputRef = useRef(null);
    const currentFolderId = path.length > 0 ? path[path.length - 1].id : null;

    const fetchItems = async () => {
        setLoading(true);
        try {
            const res = await fetch(`/api/workspace?parent_id=${currentFolderId || ''}`, { headers: { Authorization: `Bearer ${token}` } });
            const data = await res.json();
            setItems(data);
            if (data.path) setPath(data.path);
        } catch(e) {} finally { setLoading(false); }
    };

    useEffect(() => { fetchItems(); }, [currentFolderId]);

    // 处理全局拖入的文件
    useEffect(() => {
        if (filesToUpload && filesToUpload.length > 0) {
            filesToUpload.forEach(file => {
                 const fd = new FormData(); fd.append('file', file); if(currentFolderId) fd.append('folder_id', currentFolderId);
                 fetch('/api/workspace/upload', { method: 'POST', headers: { Authorization: `Bearer ${token}` }, body: fd }).then(fetchItems);
            });
            onClearUploads();
        }
    }, [filesToUpload]);

    const handleCreateFolder = async () => {
        if(!newFolderName.trim()) { setIsCreateFolder(false); return; }
        await fetch('/api/workspace/folder', { method: 'POST', headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' }, body: JSON.stringify({ name: newFolderName, parent_id: currentFolderId }) });
        setNewFolderName(""); setIsCreateFolder(false); fetchItems();
    };

    const handleUpload = async (e) => {
        const file = e.target.files[0]; if(!file) return;
        const fd = new FormData(); fd.append('file', file); if(currentFolderId) fd.append('folder_id', currentFolderId);
        await fetch('/api/workspace/upload', { method: 'POST', headers: { Authorization: `Bearer ${token}` }, body: fd });
        fetchItems();
    };

    const fmtSize = (s) => s < 1024 ? s+'B' : s < 1024*1024 ? (s/1024).toFixed(1)+'KB' : (s/1024/1024).toFixed(1)+'MB';

    return (
        <div className="flex flex-col h-full bg-white dark:bg-[#121212]">
            {/* Header */}
            <div className="px-8 py-6 flex flex-col gap-4 border-b border-gray-100 dark:border-gray-800">
                <div className="flex items-center justify-between">
                  <h1 className="text-2xl font-bold tracking-tight dark:text-white">资源库</h1>
                  <div className="flex gap-3">
                     <button onClick={() => fileInputRef.current.click()} className="flex items-center gap-2 px-4 py-2 bg-white dark:bg-[#252525] border border-gray-200 dark:border-gray-700 text-gray-700 dark:text-gray-200 rounded-lg hover:bg-gray-50 dark:hover:bg-[#333] text-sm font-medium transition-colors shadow-sm">
                       <UploadCloud className="w-4 h-4" /> 上传文件
                     </button>
                     <button onClick={() => setIsCreateFolder(true)} className="flex items-center gap-2 px-4 py-2 bg-indigo-600 text-white rounded-lg hover:bg-indigo-700 text-sm font-medium transition-colors shadow-sm shadow-indigo-500/30">
                       <Plus className="w-4 h-4" /> 新建文件夹
                     </button>
                     <input type="file" className="hidden" ref={fileInputRef} onChange={handleUpload}/>
                  </div>
                </div>
                <div className="flex items-center justify-between">
                  <div className="flex items-center gap-2 text-sm text-gray-500">
                     <Home className="w-4 h-4 cursor-pointer hover:text-indigo-600" onClick={()=>setPath([])} />
                     {path.map((p, idx) => (
                       <React.Fragment key={p.id}>
                         <ChevronRight className="w-4 h-4 text-gray-300" />
                         <span onClick={()=>{setPath(path.slice(0, idx+1))}} className={`cursor-pointer hover:text-indigo-600 ${idx === path.length - 1 ? 'font-medium text-gray-800 dark:text-gray-200' : ''}`}>{p.name}</span>
                       </React.Fragment>
                     ))}
                  </div>
                  <div className="flex items-center p-1 rounded-lg border border-gray-200 dark:border-gray-700 bg-gray-100 dark:bg-[#252525]">
                    <button onClick={() => setViewMode('grid')} className={`p-1.5 rounded-md transition-all ${viewMode === 'grid' ? 'bg-white dark:bg-[#333] shadow text-indigo-600' : 'text-gray-400'}`}><Grid className="w-4 h-4" /></button>
                    <button onClick={() => setViewMode('list')} className={`p-1.5 rounded-md transition-all ${viewMode === 'list' ? 'bg-white dark:bg-[#333] shadow text-indigo-600' : 'text-gray-400'}`}><List className="w-4 h-4" /></button>
                  </div>
                </div>
            </div>

            {/* File List */}
            <div className="flex-1 overflow-y-auto p-8">
                {isCreateFolder && (
                  <div className="mb-6 animate-in fade-in slide-in-from-top-4 duration-300">
                     <div className="p-4 rounded-xl border-2 border-indigo-500 border-dashed flex items-center gap-4 bg-indigo-50 dark:bg-indigo-900/20">
                       <FolderOpen className="w-8 h-8 text-indigo-500" />
                       <input autoFocus value={newFolderName} onChange={e=>setNewFolderName(e.target.value)} onKeyDown={e=>e.key==='Enter'&&handleCreateFolder()} placeholder="文件夹名称..." className="flex-1 bg-transparent border-none outline-none text-lg font-medium dark:text-white" />
                       <div className="flex gap-2">
                         <button onClick={() => setIsCreateFolder(false)} className="p-2 hover:bg-black/5 rounded-full"><X className="w-5 h-5 text-gray-500"/></button>
                         <button onClick={handleCreateFolder} className="p-2 bg-indigo-600 hover:bg-indigo-700 text-white rounded-full"><Check className="w-5 h-5"/></button>
                       </div>
                     </div>
                  </div>
                )}

                {loading ? <div className="flex justify-center p-10"><Loader2 className="animate-spin text-indigo-500"/></div> : (
                   <div className={viewMode === 'grid' ? "grid grid-cols-2 md:grid-cols-4 lg:grid-cols-5 gap-6" : "flex flex-col gap-2"}>
                     {items.folders.map(f => (
                       <div key={f.id} onClick={()=>setPath([...path, {id:f.id, name:f.name}])} className={`group relative border transition-all duration-200 cursor-pointer bg-white dark:bg-[#1e1e1e] border-gray-200 dark:border-gray-700 hover:border-indigo-300 hover:shadow-lg ${viewMode === 'grid' ? 'rounded-2xl p-6 flex flex-col items-center gap-4 text-center aspect-square justify-center' : 'rounded-xl p-4 flex items-center gap-4'}`}>
                         <Folder className="w-12 h-12 text-yellow-500 fill-yellow-500/20" />
                         <div className={viewMode==='grid'?'w-full':'flex-1'}><h3 className={`font-medium truncate dark:text-gray-200 ${viewMode==='grid'?'text-base':'text-sm'}`}>{f.name}</h3></div>
                       </div>

                   </div>

const ChatMessageItem = memo(({ role, content, user, streaming }) => {
    const parseContent = (text) => {
        const fileCards = [];
        let cleanText = text || '';
        const fileCardRegex = /<file-card\s+type="([^"]*)"\s+name="([^"]*)"\s+url="([^"]*)"\s*\/>/g;
        let match;
        while ((match = fileCardRegex.exec(text)) !== null) { fileCards.push({ type: match[1], name: match[2], url: match[3] }); }
        cleanText = cleanText.replace(fileCardRegex, '').trim();
        const thinkMatch = cleanText.match(/<think>([\s\S]*?)<\/think>/);
        const openThinkMatch = cleanText.match(/<think>([\s\S]*)/);
        let thought = null, answer = cleanText;
        if (thinkMatch) { thought = thinkMatch[1]; answer = cleanText.replace(thinkMatch[0], ''); }
        else if (openThinkMatch) { thought = openThinkMatch[1]; answer = ''; }
        return { thought, answer, fileCards };
    };
    const { thought, answer, fileCards } = role!=='system' ? parseContent(content) : {thought:null,answer:'',fileCards:[]};
  
    return (
      <div className={`flex gap-4 mb-6 group ${role==='user'?'flex-row-reverse':''} items-start`}>
          <div className="mt-1 shrink-0"><Avatar url={role==='user'?user.avatar:''} username={role==='user'?user.username:'AI'} size="sm"/></div>
          <div className={`p-5 rounded-2xl max-w-[85%] text-[15px] leading-7 shadow-sm ${role==='user'?'bg-indigo-600 text-white rounded-tr-sm':'bg-white dark:bg-[#1e1e1e] border border-gray-100 dark:border-gray-700 dark:text-gray-100 rounded-tl-sm'}`}>
              {thought && <ThinkingProcess content={thought} streaming={streaming && !answer} />}
              {fileCards.length > 0 && <div className="flex flex-wrap gap-2 mb-2">{fileCards.map((f,i)=><a key={i} href={f.url} target="_blank" className="flex items-center gap-2 bg-black/5 dark:bg-white/10 p-2 rounded-lg text-xs"><Paperclip size={12}/>{f.name}</a>)}</div>}
              {role==='assistant' ? <ReactMarkdown components={{code:CodeBlock}}>{answer}</ReactMarkdown> : <div className="whitespace-pre-wrap">{answer}</div>}
              {streaming && !answer && !thought && <span className="inline-block w-2 h-4 bg-indigo-500 animate-pulse rounded-full"/>}
          </div>
      </div>
    );
});

const AuthScreen = ({ onLogin }) => {
    const [isLogin, setIsLogin] = useState(true);
    const [loading, setLoading] = useState(false);
    const handleSubmit = async (e) => {
      e.preventDefault(); setLoading(true);
      const fd = new FormData(e.target);
      const endpoint = isLogin ? '/api/token' : '/api/register';
      const body = isLogin ? new URLSearchParams({ username: fd.get('username'), password: fd.get('password') }) : JSON.stringify(Object.fromEntries(fd));
      try {
        const res = await fetch(endpoint, { method: 'POST', headers: isLogin ? {'Content-Type': 'application/x-www-form-urlencoded'} : {'Content-Type': 'application/json'}, body });
        if (!res.ok) throw new Error(await res.text());
        if (isLogin) { const data = await res.json(); onLogin(data); } else { alert('注册成功'); setIsLogin(true); }
      } catch (err) { alert(err.message); } finally { setLoading(false); }
    };
    return (
      <div className="min-h-screen flex items-center justify-center bg-gray-50 dark:bg-[#050505] p-4">
        <div className="w-full max-w-sm p-10 bg-white dark:bg-[#151515] rounded-[32px] shadow-2xl border border-gray-100 dark:border-gray-800">
          <div className="text-center mb-10"><div className="w-16 h-16 bg-gradient-to-tr from-indigo-500 to-purple-600 rounded-2xl mx-auto flex items-center justify-center text-white mb-4 shadow-lg shadow-indigo-500/30"><LinkIcon size={32}/></div><h1 className="text-3xl font-bold dark:text-white">Palink AI</h1><p className="text-gray-500 text-sm mt-2">企业级智能对话助手</p></div>
          <form onSubmit={handleSubmit} className="space-y-5">
            <input name="username" placeholder="Username" className="w-full px-5 py-3 rounded-2xl bg-gray-50 dark:bg-[#202020] border-none outline-none dark:text-white" required />
            <input name="password" type="password" placeholder="Password" className="w-full px-5 py-3 rounded-2xl bg-gray-50 dark:bg-[#202020] border-none outline-none dark:text-white" required />
            <button disabled={loading} className="w-full py-3.5 bg-black dark:bg-white text-white dark:text-black rounded-2xl font-bold hover:scale-[1.02] transition-all">{loading ? <Loader2 className="animate-spin mx-auto"/> : (isLogin ? '登录' : '注册')}</button>
          </form>
          <button onClick={() => setIsLogin(!isLogin)} className="w-full mt-6 text-sm text-gray-500 hover:text-indigo-600 font-medium">{isLogin ? '创建账号' : '直接登录'}</button>
        </div>
      </div>
    );
};

  const [user, setUser] = useState(null);

  const [activeTab, setActiveTab] = useState('chat');
  const [dragActive, setDragActive] = useState(false);
  const [droppedFiles, setDroppedFiles] = useState([]);

  useEffect(() => { if(isDark) document.documentElement.classList.add('dark'); else document.documentElement.classList.remove('dark'); localStorage.setItem('theme', isDark?'dark':'light'); }, [isDark]);
  useEffect(() => { if(token) { try { const payload = JSON.parse(atob(token.split('.')[1])); fetch('/api/users/me', { headers: { Authorization: `Bearer ${token}` } }).then(r => r.ok ? r.json() : Promise.reject()).then(u => setUser({ ...u, role: payload.role || u.role })).catch(() => { setToken(null); localStorage.removeItem('palink_token'); }); } catch(e) { setToken(null); } } }, [token]);

  // Global Drag Handlers
  const handleDrag = (e) => { e.preventDefault(); e.stopPropagation(); if (e.type === 'dragenter' || e.type === 'dragover') setDragActive(true); else if (e.type === 'dragleave') setDragActive(false); };
  const handleDrop = (e) => {
      e.preventDefault(); e.stopPropagation(); setDragActive(false);
      if (e.dataTransfer.files && e.dataTransfer.files.length > 0) {
          // Pass files to active view
          setDroppedFiles(Array.from(e.dataTransfer.files));
      }
  };

  if(!token) return <AuthScreen onLogin={(d) => { setToken(d.access_token); localStorage.setItem('palink_token', d.access_token); }} />;
  if(!user) return <div className="h-screen flex items-center justify-center dark:bg-black"><Loader2 className="animate-spin text-indigo-600"/></div>;

  return (
    <div className={`flex h-screen w-full overflow-hidden transition-colors duration-300 font-sans ${isDark ? 'bg-[#0f0f0f] text-gray-100' : 'bg-white text-gray-900'}`} onDragEnter={handleDrag}>
       {/* Global Drag Overlay */}
       {dragActive && (
        <div className="absolute inset-0 z-50 flex items-center justify-center bg-indigo-500/10 backdrop-blur-sm border-4 border-dashed border-indigo-500 rounded-lg m-4" onDragEnter={handleDrag} onDragLeave={handleDrag} onDragOver={handleDrag} onDrop={handleDrop}>
          <div className="text-center animate-bounce">
            <ArrowUpCircle className="w-20 h-20 text-indigo-600 mx-auto mb-4" />
            <h2 className="text-3xl font-bold text-indigo-700">释放文件</h2>
            <p className="text-indigo-600 mt-2">松手后上传到当前 {activeTab === 'chat' ? '对话' : '文件夹'}</p>
          </div>
        </div>
      )}

       {/* Sidebar */}
       <aside className={`w-20 flex-shrink-0 flex flex-col items-center py-8 border-r ${isDark ? 'border-gray-800 bg-[#0c0c0c]' : 'border-gray-200 bg-gray-50'}`}>
         <div className="w-10 h-10 bg-indigo-600 rounded-xl flex items-center justify-center mb-10 shadow-lg shadow-indigo-500/30 text-white"><Command size={20}/></div>
         <nav className="flex-1 flex flex-col gap-6 w-full px-2">
            <NavButton active={activeTab === 'chat'} onClick={() => setActiveTab('chat')} icon={<MessageSquare />} label="对话" isDark={isDark} />
            <NavButton active={activeTab === 'workspace'} onClick={() => setActiveTab('workspace')} icon={<FolderOpen />} label="空间" isDark={isDark} />
            <NavButton active={activeTab === 'settings'} onClick={() => setActiveTab('settings')} icon={<Settings />} label="设置" isDark={isDark} />
         </nav>
         <div className="mt-auto flex flex-col gap-4 items-center mb-4">
             <Avatar url={user.avatar} size="sm" className="ring-2 ring-transparent hover:ring-indigo-500 cursor-pointer" />
         </div>
       </aside>

       {/* Main Content */}
       <main className="flex-1 relative flex flex-col h-full overflow-hidden bg-white dark:bg-[#121212]">
          {activeTab === 'chat' && <ChatView token={token} user={user} filesToUpload={droppedFiles} onClearUploads={()=>setDroppedFiles([])} />}
          {activeTab === 'workspace' && <WorkspaceView token={token} filesToUpload={droppedFiles} onClearUploads={()=>setDroppedFiles([])} />}
          {activeTab === 'settings' && <SettingsView token={token} user={user} isDark={isDark} toggleTheme={()=>setIsDark(!isDark)} onLogout={()=>{setToken(null);localStorage.removeItem('palink_token');}} />}
       </main>
    </div>
  );
}
