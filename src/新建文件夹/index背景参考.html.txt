        onBin: (buf) => {
            const v = new DataView(buf); const flag = v.getUint8(0);
            switch(flag) {
                case 0: // File Chunk
                    if(Core.role === 'sender') return;
                    Core.rbuf.push(new Uint8Array(buf).subarray(1)); 
                    Core.rsize += (buf.byteLength - 1);
                    Core.sendAck(Core.rsize); 
                    Core.upUI();
                    break;

                case 1: // ACK
                    if(Core.role === 'receiver') return;
                    const ackedSize = v.getFloat64(1);
                    if (Core.pendingAck) {
                        Core.pendingAck.resolve(true);
                        Core.pendingAck = null;
                    }
                    Core.acked = ackedSize; 
                    $('btn-kill').style.display = 'none';
                    Core.upUI();
                    break;

                case 2: // Chat
                    const txt = new TextDecoder().decode(buf.slice(1)); UI.addMsg(txt, false); 
                    break;

                case 3: // Meta Header
                    if(Core.role === 'sender') return;
                    const metaLen = v.getUint16(1);
                    try {
                        const meta = JSON.parse(new TextDecoder().decode(buf.slice(3, 3+metaLen)));
                        Core.meta = { ...(Core.meta||{}), ...meta };
                        if(!Core.rbuf || Core.rsize === 0) { Core.rsize = 0; Core.rbuf = []; Core.t0 = Date.now(); }
                        UI.go('p-trans');
                        const fileData = new Uint8Array(buf).subarray(3 + metaLen);
                        if(fileData.byteLength > 0) { Core.rbuf.push(fileData); Core.rsize += fileData.byteLength; }
                    } catch(e) {}
                    Core.sendAck(Core.rsize); 
                    Core.upUI();
                    break;
            }
        },

        startTransfer: async () => {
            if(Core.sending) return;
            Core.sending = true;
            Core.offset = 0;
            Core.acked = 0;

            $('btn-kill').style.display = 'none';

            let CHUNK_SIZE = 32 * 1024;
            if (Core.mode === 'relay') {
                 CHUNK_SIZE = 512 * 1024;
            }
            
            while(Core.offset < Core.file.size && Core.sending) {
                const end = Math.min(Core.file.size, Core.offset + CHUNK_SIZE);
                const chunkBlob = Core.file.slice(Core.offset, end);
                const chunkData = await Core.readSlice(chunkBlob);
                
                let success = false;
                let retries = 0;
                
                while(!success && Core.sending && retries < 10) {
                    try {
                        await Core.sendChunkWithAck(chunkData, Core.offset);
                        success = true;
                    } catch(e) {
                        retries++;
                        if(retries === 2 && Core.mode === 'p2p' && Core.userMode === 'auto') {
                            Core.mode = 'relay';
                            UI.setStatus('relay');
                            CHUNK_SIZE = 512 * 1024;
                            UI.toast(curLang==='cn'?'自动切换至中转...':'Switching to Relay...');
                        }
                        if(retries === 5) UI.toast(`Retry (${retries}/10)...`);
                        await new Promise(r => setTimeout(r, 1000));
                    }
                }

                if(!success) {
                    UI.err("Connection Lost");
                    return;
                }

                Core.offset = end;
                Core.upUI();
            }
            
            if(Core.sending) Core.sig({type:'done'});
            Core.sending = false;
        },

        readSlice: (blob) => {
            return new Promise((resolve, reject) => {
                const r = new FileReader();
                r.onload = e => resolve(new Uint8Array(e.target.result));
                r.onerror = reject;
                r.readAsArrayBuffer(blob);
            });
        },

        sendChunkWithAck: (chunkData, currentOffset) => {
            return new Promise((resolve, reject) => {
                let p;
                if(currentOffset === 0) {
                    const metaStr = JSON.stringify({name: Core.file.name, size: Core.file.size, type: Core.file.type});
                    const metaBytes = new TextEncoder().encode(metaStr);
                    const metaLen = metaBytes.length;
                    p = new Uint8Array(1 + 2 + metaLen + chunkData.length);
                    const v = new DataView(p.buffer);
                    p[0] = 3; 
                    v.setUint16(1, metaLen);
                    p.set(metaBytes, 3);
                    p.set(chunkData, 3 + metaLen);
                } else {
                    p = new Uint8Array(chunkData.length + 1);
                    p[0] = 0;
                    p.set(chunkData, 1);
                }

                // [FIX] Increased Timeout from 8s to 30s for P2P stability
                const timeout = setTimeout(() => {
                    Core.pendingAck = null;
                    reject('ACK Timeout');
                }, 30000);

                Core.pendingAck = {
                    resolve: (val) => { clearTimeout(timeout); resolve(val); },
                    reject: (err) => { clearTimeout(timeout); reject(err); }
                };

                try { 
                    if(Core.mode==='p2p' && Core.dc?.readyState==='open') Core.dc.send(p); 
                    else Core.ws.send(p); 
                } catch(e) {
                    clearTimeout(timeout);
                    reject('Send Error');
                }
            });
        },

        sendAck: (size) => { const b = new Uint8Array(9); b[0]=1; new DataView(b.buffer).setFloat64(1, size); Core.rawSend(b); },
        sendText: (txt) => { const t = new TextEncoder().encode(txt); const p = new Uint8Array(t.length+1); p[0]=2; p.set(t, 1); Core.rawSend(p); },
        rawSend: (data) => { try { if(Core.mode==='p2p' && Core.dc?.readyState==='open') Core.dc.send(data); else Core.ws.send(data); } catch(e){} },
        
        upUI: (force = false) => {
            const now = Date.now(); 
            if(now <= Core.t0) Core.t0 = now - 1000;
            let isFinished = false;
            if (Core.role === 'sender') isFinished = Core.file && Core.offset >= (Core.file?.size || 0);
            else isFinished = Core.meta && Core.rsize >= (Core.meta?.size || 0);
            if(!force && (now - Core.lastUI < 500) && !isFinished) return; 
            if (!Core.lastBytes) { Core.lastBytes = 0; Core.lastTime = now; }
            let cur = Core.role==='sender' ? Core.acked : Core.rsize; 
            let tot = Core.role==='sender' ? (Core.file?.size || 0) : (Core.meta?.size || 0);
            if(Core.role==='sender' && cur < Core.offset) cur = Core.offset; 
            let pct = 0;
            let infoStr = Core.mode==='p2p'?'P2P Mode':'Relay Mode';
            if (Core.userMode === 'relay') infoStr = 'Force Relay';
            else if (Core.userMode === 'p2p') infoStr = 'Force P2P';

            if (!tot && cur > 0) tot = cur; 
            if(tot > 0) { pct = (cur/tot)*100; if(isNaN(pct) || !isFinite(pct)) pct = 0; if(pct > 100) pct = 100; } 
            else if (Core.role === 'receiver') infoStr = LANG[curLang].status_ready;
            const timeDiff = (now - Core.lastTime) / 1000;
            let instSpeed = 0;
            if (timeDiff > 0) { const bytesDiff = cur - Core.lastBytes; if (bytesDiff > 0) instSpeed = bytesDiff / 1024 / 1024 / timeDiff; }
            Core.smoothSpeed = (Core.smoothSpeed || 0) * 0.7 + instSpeed * 0.3;
            if(instSpeed === 0 && timeDiff > 2) Core.smoothSpeed = 0;
            Core.lastBytes = cur; Core.lastTime = now; Core.lastUI = now;
            const spdStr = Core.smoothSpeed.toFixed(2);
            let totalStr = fmtSize(tot);
            if(tot === 0 && Core.meta?.size) totalStr = fmtSize(Core.meta.size);
            else if (tot === 0) totalStr = '?';
            let sizeStr = fmtSize(cur) + ' / ' + totalStr;
            let etaStr = '--:--';
            if(Core.smoothSpeed > 0 && tot > 0) { const remain = Math.max(0, tot - cur); const sec = remain / (Core.smoothSpeed * 1024 * 1024); etaStr = fmtTime(sec); }
            if(pct >= 100) etaStr = '00:00';
            UI.update(pct, spdStr+' MB/s', infoStr, sizeStr, etaStr);
            if(pct >= 100 && tot > 0 && Core.role==='sender') { Core.sending = false; Core.sig({type:'done'}); }
        },

        create: () => {
            Core.role = 'sender'; Core.room = Math.random().toString(36).substr(2,4).toUpperCase();
            const storageKey = `quickshare_${Date.now()}_${Math.random()}`; sessionStorage.setItem(storageKey, 'sender');
            Object.keys(sessionStorage).forEach(key => { if (key.startsWith('quickshare_') && key !== storageKey) { try { if (Date.now() - parseInt(key.split('_')[1]) > 30000) sessionStorage.removeItem(key); } catch(e) {} } });
            const newUrl = location.protocol+'//'+location.host+location.pathname+'?c='+Core.room;
            window.history.pushState({path:newUrl},'',newUrl);
            Core.sig({type:'join', role:'sender'}); $('code-show').innerText = Core.room;
            $('qr').innerHTML=''; try { new QRCode($('qr'), {text:newUrl, width:160, height:160}); } catch(e){ alert('二维码生成失败'); }
            UI.go('p-wait');
        },
        join: () => {
            const c = $('c-in').value.toUpperCase().replace(/[^A-Z0-9]/g, '');
            if(c.length!==4) return alert('Code Error');
            const storageKey = `quickshare_${Date.now()}_${Math.random()}`; sessionStorage.setItem(storageKey, 'receiver');
            Core.role = 'receiver'; Core.room = c; Core.sig({type:'join', role:'receiver'}); UI.go('p-trans'); UI.setStatus('connecting');
        },
        setupP2P: () => {
            if (Core.userMode === 'relay') return; 

            log('P2P Start'); const pc = new RTCPeerConnection(Core.rtcConfig); Core.peer = pc;
            
            // 增加ICE超时时间，确保充分尝试所有ICE服务器（STUN先，然后TURN）
            const iceTimeout = setTimeout(() => {
                if (pc.connectionState !== 'connected' && pc.connectionState !== 'connecting') {
                    log('P2P Timeout -> Relay');
                    if (Core.peer === pc) { 
                        if (Core.userMode === 'p2p') {
                             UI.err('P2P Connection Timeout');
                             pc.close(); Core.peer = null;
                             return;
                        }
                        pc.close(); Core.peer = null; UI.toast('P2P Timeout, using Server Relay');
                        Core.mode = 'relay'; UI.setStatus('relay'); Core.sig({type:'fallback-relay'});
                    }
                }
            }, 20000); // 增加到20秒，确保充分尝试所有STUN和TURN服务器
            
            // 监听ICE连接状态变化，确保完全尝试所有ICE服务器
            pc.oniceconnectionstatechange = () => {
                log('ICE Connection State:', pc.iceConnectionState);
                if (pc.iceConnectionState === 'disconnected' || pc.iceConnectionState === 'failed' || pc.iceConnectionState === 'closed') {
                    clearTimeout(iceTimeout);
                    if (Core.peer === pc) {
                        if (Core.userMode === 'p2p') {
                            UI.err('P2P Connection Failed');
                            pc.close(); Core.peer = null;
                            return;
                        }
                        UI.toast('P2P Blocked, using Server Relay');
                        Core.mode = 'relay';
                        UI.setStatus('relay');
                        Core.sig({type:'fallback-relay'});
                    }
                }
            };
            
            pc.onconnectionstatechange = () => { 
                clearTimeout(iceTimeout);
                if(pc.connectionState==='connected') { 
                    Core.mode = 'p2p'; UI.setStatus('p2p'); Core.sig({type:'log-p2p'}); 
                }
                else if (pc.connectionState === 'failed') {
                    if(Core.mode === 'p2p') { 
                        if (Core.userMode === 'p2p') { UI.err('P2P Connection Failed'); return; }
                        UI.toast('P2P Blocked, using Server Relay'); 
                        Core.mode = 'relay'; 
                        UI.setStatus('relay'); 
                        Core.sig({type:'fallback-relay'}); 
                    }
                }
            };
            
            // 确保ICE候选收集完成后再决定是否切换到中转
            pc.onicecandidate = e => { 
                if(e.candidate) {
                    Core.sig({type:'signal', payload:{candidate:e.candidate}});
                } else {
                    log('ICE Candidates Gathering Complete');
                    // ICE候选收集完成，但不立即关闭超时，继续等待连接建立
                }
            };
            
            const dc = pc.createDataChannel('file', {ordered:true});
            dc.onopen = () => { Core.mode='p2p'; UI.setStatus('p2p'); Core.dc=dc; Core.sig({type:'log-p2p'}); };
            dc.onmessage = e => Core.onBin(e.data);
            pc.createOffer().then(o => pc.setLocalDescription(o)).then(() => Core.sig({type:'signal', payload:{sdp:pc.localDescription}}));
        },
        onPeerSig: async (d) => {
            if(!Core.peer) { 
                const pc = new RTCPeerConnection(Core.rtcConfig); Core.peer = pc;
                pc.ondatachannel = e => { Core.dc = e.channel; Core.dc.onmessage = ev => Core.onBin(ev.data); Core.mode = 'p2p'; UI.setStatus('p2p'); Core.sig({type:'log-p2p'}); };
                pc.onicecandidate = e => e.candidate && Core.sig({type:'signal', payload:{candidate:e.candidate}});
            }
            if(d.sdp) { await Core.peer.setRemoteDescription(d.sdp); if(d.sdp.type==='offer') { const a = await Core.peer.createAnswer(); await Core.peer.setLocalDescription(a); Core.sig({type:'signal', payload:{sdp:a}}); } } 
            else if(d.candidate) Core.peer.addIceCandidate(d.candidate);
        },
        save: () => { try { const b = new Blob(Core.rbuf, {type:Core.meta.type}); if (Core.meta && Core.meta.size && b.size !== Core.meta.size) { UI.toast(`Size Mismatch`); const u = URL.createObjectURL(b); const a = document.createElement('a'); a.href=u; a.download=`(incomplete)_${Core.meta.name}`; a.click(); return; } const u = URL.createObjectURL(b); const a = document.createElement('a'); a.href=u; a.download=Core.meta.name; a.click(); } catch(e) { UI.err('Memory Error / Save Failed'); } Core.rbuf=[]; },
        copy: () => { const link = location.protocol+'//'+location.host+location.pathname+'?c='+Core.room; navigator.clipboard.writeText(link); UI.toast('Link Copied'); }
    };

    window.UI = UI; window.Core = Core; document.querySelectorAll('.kinetic-btn').forEach(el => new KineticBtn(el)); Core.init();
})();
